Composition root

Assets/_Project/Scripts/Session/GameplayComposer.cs

New Game is not a scatter of FindObjectOfType calls. GameplayComposer manually constructs World, Fog, Pathfinding, Economy, Tasks, Combat, Agents, Labor, and Buildings, then late-binds displacement so doors and walls can eject units. That is how a designated dig becomes a claimed task on a living dwarf.

    /// <summary>
    /// Manual composition root (ADR-0005). Concrete construction of domain services lives only here.
    /// </summary>
    public sealed class GameplayComposer : IDisposable
    {
        public GameplayComposer(TerrainCatalog terrainCatalog, ResourceCatalog resourceCatalog = null)
        {
            // ...
            EventBus = new EventBus();
            Clock = new SimulationClock();
            MotionSpace = new MotionSpaceState();
            Settings = new SettingsService();
            Settings.Load();

            World = new WorldService(
                EventBus,
                TerrainCatalog,
                WorldConstants.DefaultWidth,
                WorldConstants.DefaultHeight);

            Fog = new FogService(WorldConstants.DefaultWidth, WorldConstants.DefaultHeight);

            // Colonist pathfinding needs buildings for floors/doors; bind Structures after Buildings.
            ColonistWalk = new ColonistWalkabilityMap(World);
            Pathfinding = new PathfindingService(ColonistWalk, World, World);

            Economy = new EconomyService(EventBus);
            Combat = new CombatService(EventBus);
            Tasks = new TaskService(EventBus, Pathfinding, World, MotionSpace);
            // ...
            Buildings = new BuildingService(
                EventBus,
                World,
                Economy,
                Tasks,
                BuildingAccess,
                Stockpile,
                BuildingsCatalog,
                () => Pathfinding.InvalidatePathCache(),
                tiles => displacementHolder?.EjectFromTiles(tiles),
                tiles =>
                {
                    agentsHolder?.ClearPathsThroughTiles(tiles);
                    enemiesHolder?.ClearPathsThroughTiles(tiles);
                });
            Agents = new AgentService(
                Buildings, World, Stockpile, EventBus, new DwarfNamePool(),
                Fog, Combat, Tasks, MotionSpace);
            Enemies = new EnemyService(
                EventBus, World, Pathfinding, Agents, Combat,
                Buildings as IStructurePassabilityQuery, MotionSpace);
            Labor = new LaborService(
                Agents, Tasks, World, Pathfinding, Economy, Stockpile,
                Buildings, ResourceCatalog, Enemies, MotionSpace);
            // ...
        }
    }

Typed event bus

Assets/_Project/Scripts/Core/EventBus.cs

Events are structs. Publish is synchronous. Handlers are snapshotted so a subscriber may unsubscribe mid-broadcast without corrupting the list. Building constructed, dwarf died, stock changed — the hall talks to itself without a god object.

        public void Publish<T>(T evt) where T : struct
        {
            if (!_handlers.TryGetValue(typeof(T), out var list) || list.Count == 0)
            {
                return;
            }

            // Snapshot so handlers may subscribe/unsubscribe during publish.
            var snapshot = list.ToArray();
            for (var i = 0; i < snapshot.Length; i++)
            {
                if (snapshot[i] is Action<T> action)
                {
                    action.Invoke(evt);
                }
            }
        }

        public IDisposable Subscribe<T>(Action<T> handler) where T : struct
        {
            if (handler == null)
            {
                throw new ArgumentNullException(nameof(handler));
            }

            var type = typeof(T);
            if (!_handlers.TryGetValue(type, out var list))
            {
                list = new List<Delegate>();
                _handlers[type] = list;
            }

            list.Add(handler);
            return new Subscription(() => Unsubscribe(type, handler));
        }

Volume A*

Assets/_Project/Scripts/Pathfinding/VolumeAStarPathfinder.cs

Dwarves walk XYZ. Neighbors are ±X ±Z on a floor, and ±Y on a ladder live-stand. The search is capped at 1000 expansions — CapExceeded is a real status, not an infinite crawl through the cave. Designate a cell; this is how a dwarf finds a legal stand beside it.

    /// </summary>

        public VolumePathResult FindPath(Vector3Int start, Vector3Int goal)
        {
            if (_volume == null)
            {
                return VolumePathResult.Failure(PathStatus.OutOfBounds);
            }

            var startIn = _volume.InBounds(start.x, start.y, start.z);
            var goalIn = _volume.InBounds(goal.x, goal.y, goal.z);
            if (!startIn || !goalIn)
            {
                return VolumePathResult.Failure(PathStatus.OutOfBounds);
            }

            if (!VolumeLiveStep.CanVisit(
                    _volume, _walkability, _ladders, _planThrough, start.x, start.y, start.z))
            {
                return VolumePathResult.Failure(PathStatus.NonWalkableStart);
            }

                    _volume, _walkability, _ladders, _planThrough, goal.x, goal.y, goal.z))
            {
                return VolumePathResult.Failure(PathStatus.NonWalkableGoal);
            }

            if (start == goal)
            {
                return VolumePathResult.Success(new[] { start });
            }
            // ...
            return Search(start, goal);
        }

                    return VolumePathResult.Failure(PathStatus.CapExceeded);

Combat kernel

Assets/_Project/Scripts/Combat/DamageCalculator.cs

Actual damage is max(0, raw − defense). Health never goes below zero. A Giant Rat on the stockpile and an Ork at the door use the same pure function. Starvation is a separate apply that ignores defense — that lock lived in the Food brief, not here.

    /// <summary>
    /// Pure damage rules. Actual = max(0, raw − defense); health clamped to [0, max].
    /// </summary>
    public static class DamageCalculator
    {
        public static int ComputeActualDamage(int rawDamage, int defense)
        {
            if (rawDamage <= 0)
            {
                return 0;
            }

            if (defense < 0)
            {
                defense = 0;
            }

            var actual = rawDamage - defense;
            return actual < 0 ? 0 : actual;
        }

        public static DamageResult Apply(int currentHealth, int maxHealth, int defense, int rawDamage)
        {
            // ... clamps currentHealth to [0, maxHealth]
            var actual = ComputeActualDamage(rawDamage, defense);
            var remaining = currentHealth - actual;
            if (remaining < 0)
            {
                remaining = 0;
            }

            // Lethal when HP reaches 0 (including already-dead; callers treat Kill as idempotent).
            return new DamageResult(actual, remaining, remaining <= 0);
        }
    }

Fog of the unexplored

Assets/_Project/Scripts/Fog/FogService.cs

The cavern stays dark until a dwarf walks or a fireplace burns. Visibility is max-merge on Chebyshev rings; already-satisfied rings are skipped; only dirty cells wake the overlay. That is firelight against cave dusk as a grid, not as a filter.

        public void ApplyExplore(Vector2Int origin) =>
            ApplyRing(origin, FogConstants.DwarfFullRadius, FogConstants.DwarfSoftRadius);

        public void ApplyFireplaceLight(Vector2Int origin) =>
            ApplyRing(origin, FogConstants.FireplaceFullRadius, FogConstants.FireplaceSoftRadius);

        private void ApplyRing(Vector2Int origin, int fullRadius, int softRadius)
        {
            if (!InBounds(origin.x, origin.y))
            {
                return;
            }

            // D: skip entire ring when every cell already meets its target visibility.
            if (RingAlreadySatisfied(origin, fullRadius, softRadius))
            {
                return;
            }

            var changed = false;
            var r = softRadius;
            for (var dy = -r; dy <= r; dy++)
            {
                for (var dx = -r; dx <= r; dx++)
                {
                    var x = origin.x + dx;
                    var y = origin.y + dy;
                    if (!InBounds(x, y))
                    {
                        continue;
                    }

                    var d = Math.Max(Math.Abs(dx), Math.Abs(dy));
                    float v;
                    if (d <= fullRadius)
                    {
                        v = FogConstants.FullVisibility;
                    }
                    else if (d <= softRadius)
                    {
                        v = FogConstants.SoftVisibility;
                    }
                    else
                    {
                        continue;
                    }

                    var old = _visibility[x, y];
                    if (v > old)
                    {
                        _visibility[x, y] = v;
                        MarkDirty(x, y);
                        changed = true;
                    }
                }
            }
            // ... Generation++ if changed
        }
